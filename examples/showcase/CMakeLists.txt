# showcase: WGT's demo on Esia (its panels, the dock, the island, the status bar), in the frame of glass_window
# (glass_app). Its captures were compared with WGT's reference screenshots (docs/UI_WIDGETS.md, section 11).
if(NOT TARGET glass_app)
    message(STATUS "showcase: needs glass_window's frame (Win32, X11 or Android + a window-capable backend, or macOS + Metal) - not built")
    return()
endif()
if(APPLE)
    enable_language(OBJCXX)   # per directory: the rules for .mm files
    add_executable(showcase main.cpp image_file_apple.mm image_file.hpp)
    set_source_files_properties(image_file_apple.mm PROPERTIES COMPILE_OPTIONS "${ESIA_METAL_OBJC_OPTIONS}")
    target_link_libraries(showcase PRIVATE glass_app esia::ui "-framework ImageIO" "-framework CoreGraphics" "-framework Foundation")
    # iOS: the wallpaper is one of the building Mac's (iOS has none an app can read), a JPEG of 2560 pixels in the bundle
    set(_wallpaper)
    if(IOS)
        foreach(_source "/System/Library/Desktop Pictures/Sonoma.heic" "/System/Library/Desktop Pictures/Mac Blue.heic")
            if(EXISTS "${_source}")
                set(_wallpaper ${CMAKE_CURRENT_BINARY_DIR}/wallpaper.jpg)
                add_custom_command(OUTPUT ${_wallpaper}
                                   COMMAND sips -s format jpeg -s formatOptions 90 -Z 2560 ${_source} --out ${_wallpaper}
                                   DEPENDS ${_source} VERBATIM)
                target_sources(showcase PRIVATE ${_wallpaper})
                break()
            endif()
        endforeach()
    endif()
    glass_ios_app(showcase "Esia Showcase" showcase RESOURCES ${_wallpaper})
elseif(WIN32)
    add_executable(showcase main.cpp image_file.cpp image_file.hpp)
    target_link_libraries(showcase PRIVATE glass_app esia::ui windowscodecs ole32)
else()
    # Android: the wallpaper is one of the building machine's (an app may read only the system's built-in one), packed
    # into the APK (assets/wallpaper.*: Android's BitmapFactory decodes it); ESIA_ANDROID_WALLPAPER names another
    if(ANDROID)
        set(ESIA_ANDROID_WALLPAPER "" CACHE FILEPATH "Android: the showcase's wallpaper (empty: the building machine's)")
        set(_source "${ESIA_ANDROID_WALLPAPER}")
        file(TO_CMAKE_PATH "$ENV{SystemRoot}" _windows)
        foreach(_candidate "${_windows}/Web/Wallpaper/Windows/img0.jpg" "/System/Library/Desktop Pictures/Sonoma.heic"
                           "/usr/share/backgrounds/warty-final-ubuntu.png" "/usr/share/backgrounds/gnome/adwaita-l.jpg")
            if(NOT _source AND EXISTS "${_candidate}")
                set(_source "${_candidate}")
            endif()
        endforeach()
        if(_source)
            get_filename_component(_ext "${_source}" LAST_EXT)
            string(TOLOWER "${_ext}" _ext)
            set(GLASS_APP_ASSETS ${CMAKE_CURRENT_BINARY_DIR}/android-assets/wallpaper${_ext})
            configure_file("${_source}" ${GLASS_APP_ASSETS} COPYONLY)
        endif()
    endif()
    # Linux and Android: PNG wallpapers through the test kit (glass_app links it), JPEG ones when libjpeg is installed
    glass_add_app(showcase main.cpp image_file_linux.cpp image_file.hpp)
    target_link_libraries(showcase PRIVATE glass_app esia::ui)
    find_package(JPEG)
    if(TARGET JPEG::JPEG)
        target_link_libraries(showcase PRIVATE JPEG::JPEG)
        target_compile_definitions(showcase PRIVATE SHOWCASE_JPEG)
    endif()
endif()
esia_target_defaults(showcase)
