# glass_window: Esia in a real window - the Win32 platform layer with every window-capable backend built into this
# tree (--api picks one), liquid glass through Painter, and the core's windows. A smoke test of the whole path, not the
# demo (docs/PLATFORM_WIN32.md).
# macOS / iOS: the frame on AppKit or UIKit and the Metal backend (app_macos.mm, app_ios.mm on app_apple.mm) instead of
# the Win32 layer and its hosts. On iOS the examples are app bundles (glass_ios_app), signed after the link when the Mac
# has a provisioning profile for them (tools/ios/codesign.py).
# Linux: the frame on X11 (app_linux.cpp) with OpenGL through EGL and Vulkan through VK_KHR_xlib_surface.
# Android: a NativeActivity (app_android.cpp) with OpenGL ES through EGL and Vulkan; the examples are shared libraries
# packaged as APKs (glass_add_app, tools/android/package.py).
if(IOS)
    set(ESIA_IOS_BUNDLE_PREFIX "org.esia" CACHE STRING "iOS: the examples' bundle identifiers are <prefix>.<example>")
    set(ESIA_IOS_PROVISIONING_PROFILE "" CACHE FILEPATH "iOS: the profile the examples are signed with (default: the best one Xcode has)")
    option(ESIA_IOS_CODESIGN "iOS: sign the examples' app bundles after the link" ON)
endif()

# glass_ios_app(<target> <display name> <identifier> [RESOURCES files...]): on iOS the target becomes an app bundle
# (Info-ios.plist.in, <prefix>.<identifier>) with the files at its top level, signed last; elsewhere nothing.
function(glass_ios_app target name identifier)
    if(NOT IOS)
        return()
    endif()
    cmake_parse_arguments(PARSE_ARGV 3 arg "" "" "RESOURCES")
    set_target_properties(${target} PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/Info-ios.plist.in
        MACOSX_BUNDLE_BUNDLE_NAME "${name}"
        MACOSX_BUNDLE_GUI_IDENTIFIER ${ESIA_IOS_BUNDLE_PREFIX}.${identifier}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION})
    foreach(_file ${arg_RESOURCES})
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${_file} $<TARGET_BUNDLE_DIR:${target}>/ VERBATIM)
    endforeach()
    if(ESIA_IOS_CODESIGN)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        set(_profile)
        if(ESIA_IOS_PROVISIONING_PROFILE)
            set(_profile --profile ${ESIA_IOS_PROVISIONING_PROFILE})
        endif()
        add_custom_command(TARGET ${target} POST_BUILD
                           COMMAND ${Python3_EXECUTABLE} ${ESIA_ROOT}/tools/ios/codesign.py $<TARGET_BUNDLE_DIR:${target}> ${_profile} VERBATIM)
    endif()
endfunction()

# glass_add_app(<target> sources...): an example - an executable, on Android a shared library lib<target>.so whose main
# the frame calls (renamed glass_main), packaged as build/.../apk/<target>.apk when the Android SDK is known
# (ESIA_ANDROID_SDK, by default ANDROID_HOME or ANDROID_SDK_ROOT), with the icon font among its assets.
if(ANDROID)
    set(_sdk "$ENV{ANDROID_HOME}")
    if(NOT _sdk)
        set(_sdk "$ENV{ANDROID_SDK_ROOT}")
    endif()
    set(ESIA_ANDROID_SDK "${_sdk}" CACHE PATH "Android: the SDK (platforms, build-tools) the examples' APKs are made with")
    set(ESIA_ANDROID_PACKAGE_PREFIX "org.esia" CACHE STRING "Android: the examples' packages are <prefix>.<example>")

    # The icons (icons_material.hpp): Google's Material Icons Outlined (Apache License 2.0), downloaded once into the
    # build tree. Offline: ESIA_ANDROID_ICON_FONT names a copy. Without it the examples run without icons.
    set(ESIA_ANDROID_ICON_FONT "" CACHE FILEPATH "Android: MaterialIconsOutlined-Regular.otf for the examples' icons (empty: downloaded)")
    set(_icons "${ESIA_ANDROID_ICON_FONT}")
    if(NOT _icons)
        set(_icons ${CMAKE_BINARY_DIR}/_deps/material-icons/MaterialIconsOutlined-Regular.otf)
        set(_sha256 36370d0b1657b2e3c1f905b8966e99e6ef41690145130fe08f0706582c551e7f)
        if(NOT EXISTS ${_icons})
            set(_part ${_icons}.part)
            file(DOWNLOAD https://raw.githubusercontent.com/google/material-design-icons/4.0.0/font/MaterialIconsOutlined-Regular.otf
                 ${_part} STATUS _status)
            list(GET _status 0 _code)
            set(_hash "")
            if(_code EQUAL 0)
                file(SHA256 ${_part} _hash)
            endif()
            if(_hash STREQUAL _sha256)
                file(RENAME ${_part} ${_icons})
            else()
                file(REMOVE ${_part})
                message(WARNING "glass_window: Material Icons could not be downloaded (${_status}): the examples draw no icons "
                                "(ESIA_ANDROID_ICON_FONT names a local copy)")
                set(_icons "")
            endif()
        endif()
    endif()
    set_property(GLOBAL PROPERTY GLASS_ANDROID_ASSETS ${_icons})
endif()
function(glass_add_app target)
    if(NOT ANDROID)
        add_executable(${target} ${ARGN})
        return()
    endif()
    add_library(${target} SHARED ${ARGN})
    target_compile_definitions(${target} PRIVATE main=glass_main)
    if(ESIA_ANDROID_SDK AND EXISTS "${ESIA_ANDROID_SDK}/build-tools")
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        set(_apk ${CMAKE_BINARY_DIR}/apk/${target}.apk)
        string(REPLACE "_" "-" _pkg ${target})
        string(REGEX REPLACE "^android-" "" _min ${ANDROID_PLATFORM})
        get_property(_assets GLOBAL PROPERTY GLASS_ANDROID_ASSETS)
        set(_assetArgs)
        foreach(_asset ${_assets})
            list(APPEND _assetArgs --asset ${_asset})
        endforeach()
        add_custom_command(OUTPUT ${_apk}
                           COMMAND ${Python3_EXECUTABLE} ${ESIA_ROOT}/tools/android/package.py --lib $<TARGET_FILE:${target}> --name ${target}
                                   --package ${ESIA_ANDROID_PACKAGE_PREFIX}.${target} --label "Esia ${target}" --sdk ${ESIA_ANDROID_SDK}
                                   --abi ${ANDROID_ABI} --min-sdk ${_min} --out ${_apk} ${_assetArgs}
                           DEPENDS ${target} ${ESIA_ROOT}/tools/android/package.py ${_assets} VERBATIM)
        add_custom_target(${target}_apk ALL DEPENDS ${_apk})
    endif()
endfunction()
