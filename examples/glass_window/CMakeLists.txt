# glass_window: Esia in a real window - the Win32 platform layer with every window-capable backend built into this
# tree (--api picks one), liquid glass through Painter, and the core's windows. A smoke test of the whole path, not the
# demo (docs/PLATFORM_WIN32.md).
if(NOT TARGET esia_platform_win32)
    message(STATUS "glass_window: needs the Win32 platform layer (ESIA_PLATFORM_WIN32) - not built")
    return()
endif()

# one host file per API whose backend is built: GLASS_<API> tells host.cpp which ones exist
set(_sources main.cpp host.cpp scene.cpp ${ESIA_ROOT}/tests/support/image.cpp)   # the test kit's PNG writer
set(_libs)
set(_defs)
if(TARGET esia_rhi_d3d10 OR TARGET esia_rhi_d3d11 OR TARGET esia_rhi_d3d12)
    list(APPEND _sources dxgi_swap_chain.cpp)
    list(APPEND _libs dxgi)
endif()
foreach(_api d3d9 d3d10 d3d11 d3d12 opengl vulkan)
    if(TARGET esia_rhi_${_api})
        string(TOUPPER ${_api} _API)
        list(APPEND _sources host_${_api}.cpp)
        list(APPEND _libs esia_rhi_${_api})
        list(APPEND _defs GLASS_${_API})
    endif()
endforeach()
if(NOT _defs)
    message(STATUS "glass_window: no window-capable backend is built (ESIA_BACKEND_D3D9 / D3D10 / D3D11 / D3D12 / OPENGL / "
                   "VULKAN) - not built")
    return()
endif()
if(TARGET esia_rhi_d3d9)
    list(APPEND _libs d3d9)
endif()
if(TARGET esia_rhi_d3d10)
    list(APPEND _libs d3d10)
endif()
if(TARGET esia_rhi_d3d11)
    list(APPEND _libs d3d11)
endif()
if(TARGET esia_rhi_d3d12)
    list(APPEND _libs d3d12)
endif()
if(TARGET esia_rhi_opengl)
    list(APPEND _libs opengl32 gdi32)
endif()
if(TARGET esia_text_ft)
    list(APPEND _libs esia_text_ft)
    list(APPEND _defs GLASS_TEXT)
endif()

add_executable(glass_window ${_sources})
target_include_directories(glass_window PRIVATE ${ESIA_ROOT}/tests/support)
target_link_libraries(glass_window PRIVATE esia::render esia_platform_win32 ${_libs})
target_compile_definitions(glass_window PRIVATE ${_defs} UNICODE _UNICODE _WIN32_WINNT=0x0A00)
esia_target_defaults(glass_window)
if(MINGW AND TARGET esia_rhi_d3d_common)
    # d3d_util.hpp's std::make_shared defines std::type_info::operator== next to libstdc++ 13's own (mingw-w64 only,
    # docs/backends/README.md section 6): the two are the same inline function, so the linker may keep either
    target_link_options(glass_window PRIVATE -Wl,--allow-multiple-definition)
endif()
message(STATUS "glass_window: ${_defs}")
